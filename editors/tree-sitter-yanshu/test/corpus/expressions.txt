(program
  (name expressions)
  (version 3)
  (capabilities log)
  (data outcome
    (accepted value)
    (denied reason))
  (def decide
    (fn (amount)
      (let ((positive (and (> amount 0) (< amount 1000)))
            (decision
              (cond
                (positive (accepted amount))
                ((or (= amount 0) (< amount 0)) (denied "non-positive"))
                (else (denied "too large")))))
        (match decision
          ((accepted value) (do (log value) value))
          ((denied reason) reason)
          (_ "invalid")))))
  (export decide))
